# Comb's cases of the GitHub login in scripts/check-devcontainer.sh (DEV-gh-login-outlives-rebuild),
# read by scripts/test-check-devcontainer.sh.
#
# The format is that of scripts/fixtures/check-devcontainer.txt, its steps included, with three
# changes. The base tree adds to devcontainer.json, before the postStartCommand, the line
#
#   "mounts": [ "source=${localEnv:HOME}${localEnv:USERPROFILE}/.claude,target=/home/vscode/.claude,type=bind", "source=${localEnv:HOME}${localEnv:USERPROFILE}/.config/gh,target=/home/vscode/.config/gh,type=bind" ],
#
# and the line
#
#   "initializeCommand": "(mkdir \"${localEnv:HOME}${localEnv:USERPROFILE}/.config\" || cd .) && (mkdir \"${localEnv:HOME}${localEnv:USERPROFILE}/.config/gh\" || cd .) && exit 0",
#
# and its <post> adds, after the index entry, the entry
#
#   "credentials": "gh auth setup-git --hostname github.com --force"
#
# Two steps are added, each of which writes devcontainer.json again like extra, post, and feature:
#
#   mounts [json]               the json as the value of mounts, or no such line when no json is
#                               given
#   init [json]                 the json as the value of initializeCommand, or no such line when
#                               no json is given
#
# A post step writes the json as it stands; it does not add the credentials entry.
